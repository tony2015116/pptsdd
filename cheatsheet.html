pak::pkg_install("tony2015116/tidypptsdd")
library(tidypptsdd)
# ~/.Renviron: NEDAP_URL, NEDAP_USERNAME,
# NEDAP_PASSWORD; restart R afterwards
url <- Sys.getenv("NEDAP_URL")
username <- Sys.getenv("NEDAP_USERNAME")
password <- Sys.getenv("NEDAP_PASSWORD")
download_path <- "D:/nedap-csv"
Windows and R ≥ 4.1.0. Browser downloads need Chrome or Edge and wdauto ≥ 0.2.0; Java is not needed.

HTTP downloads no browser

get_csv(url, username, password, location_min, location_max, date_min, date_max, download_path, show_message = FALSE)

One CSV for a range of stations and dates, saved in download_path.

get_csv_auto(url, username, password, location, date, download_path, show_message = FALSE)

One CSV per station and day, filed into NonEmptyCSVs/ or EmptyCSVs/.

get_csv_auto(
  url, username, password,
  location = c("606", "608"),
  date = seq(as.Date("2024-02-18"),
             as.Date("2024-02-19"),
             by = "day"),
  download_path = download_path)
Start here when the default CSV export is the report you need.

Browser downloads Chrome or Edge

connect_to_browser(dest_dir, browser = "chrome")

Downloads the driver and starts the WebDriver service through wdauto.

download_csv_auto(url, username, password, csv_position, location, date, download_path, browser = "chrome")

csv_position is the position of the report in the website's CSV list, as a string ("1").

options(wdauto.mirror = "npmmirror")  # China
connect_to_browser("C:/webdriver")
download_csv_auto(
  url, username, password,
  csv_position = "1",
  location = c("606", "608"),
  date = "2024-02-18",
  download_path = download_path)
disconnect_from_browser(browser = "chrome")

Stops the service. For Edge pass browser = "edge" to all three functions.

Daily schedule yesterday's data

get_csv_schedule(taskname, schedule, starttime, startdate, rscript_args = NULL, ...)
download_csv_schedule(taskname, schedule, starttime, startdate, rscript_args = NULL, ...)

Registers a Windows task that runs get_csv_auto() / download_csv_auto() for the previous day.

rscript_argsContent
requiredurl, username, location, download_path
passwordpassword_env = "NEDAP_PASSWORD", the variable name
browser onlycsv_position; optional browser
get_csv_schedule(
  taskname = "nedap_csv_daily",
  schedule = "DAILY",
  starttime = "20:00",
  startdate = format(Sys.Date() + 1,
                     "%Y/%m/%d"),
  rscript_args = list(
    url = url, username = username,
    password_env = "NEDAP_PASSWORD",
    location = c("606", "608"),
    download_path = download_path))
  • startdate uses the Windows regional date format
  • Scripts and logs: C:/Temp/tidypptsdd/
# delete the task
taskscheduleR::taskscheduler_delete(
  "nedap_csv_daily")

Output files one report per station and day

download_path/
  NonEmptyCSVs/606/*.csv  # with data rows
  EmptyCSVs/608/*.csv     # header only

File names contain the report date and the download date: location606_2024-02-18data_2024-02-20download.csv.

get_csv() saves its single report directly in download_path.
